        .submit-btn:disabled { background: #a5c3e8; cursor: not-allowed; }

<div class="container">

    <h3>Выберите дату записи</h3>

    <div id="calendar"></div>

    

    <h4>Доступное время:</h4>

    <div id="slots" class="time-slots">Загрузка...</div>

    

    <button class="submit-btn" id="bookBtn" disabled>Подтвердить бронь</button>

</div>


<script>

    vkBridge.send('VKWebAppInit');


    // ⚠️ УКАЖИТЕ АДРЕС ВАШЕГО СЕРВЕРА n8n

    const N8N_URL = 'https://your-n8n-domain.com';


    let selectedDate = null;

    let selectedTime = null;

    let availableSlotsData = {}; 


    // Извлекаем переданные ботом параметры из hash (например, #staff_id=123&service_id=456)

    const hashParams = new URLSearchParams(window.location.hash.substring(1));

    const staffId = hashParams.get('staff_id');

    const serviceId = hashParams.get('service_id');


    const slotsContainer = document.getElementById('slots');


    // 1. Запрашиваем у n8n расписание под конкретного мастера и услугу

    fetch(`${N8N_URL}/get-slots?staff_id=${staffId}&service_id=${serviceId}`)

        .then(res => res.json())

        .then(data => {

            availableSlotsData = data; // Ожидаем формат: {"2026-10-25": ["10:00", "12:30"], "2026-10-26": ["09:00"]}

            slotsContainer.innerHTML = 'Выберите день на календаре выше';

            initCalendar();

        })

        .catch(() => {

            slotsContainer.innerHTML = '<div style="color:red">Ошибка загрузки дат</div>';

        });


    // 2. Инициализация календаря

    function initCalendar() {

        new AirDatepicker('#calendar', {

            inline: true,

            minDate: new Date(),

            onRenderCell({date, cellType}) {

                if (cellType === 'day') {

                    const localDateStr = formatDate(date);

                    // Если даты нет в ответе n8n — блокируем её

                    const isAvailable = availableSlotsData[localDateStr] && availableSlotsData[localDateStr].length > 0;

                    return { disabled: !isAvailable }

                }

            },

            onSelect({date}) {

                if (!date) return;

                selectedDate = formatDate(date);

                renderTimeSlots(availableSlotsData[selectedDate]);

            }

        });

    }


    // 3. Вывод кнопок времени

    function renderTimeSlots(slots) {

        slotsContainer.innerHTML = '';

        if (!slots || slots.length === 0) {

            slotsContainer.innerHTML = 'Нет свободного времени';

            return;

        }

        slots.forEach(time => {

            const btn = document.createElement('button');

            btn.className = 'time-btn';

            btn.innerText = time;

            btn.onclick = () => {

                document.querySelectorAll('.time-btn').forEach(b => b.classList.remove('active'));

                btn.classList.add('active');

                selectedTime = time;

                document.getElementById('bookBtn').disabled = false;

            };

            slotsContainer.appendChild(btn);

        });

    }


    function formatDate(date) {

        const offset = date.getTimezoneOffset();

        const localDate = new Date(date.getTime() - (offset * 60 * 1000));

        return localDate.toISOString().split('T')[0];

    }


    // 4. Отправка итоговой записи в n8n

    document.getElementById('bookBtn').onclick = async () => {

        document.getElementById('bookBtn').disabled = true;

        document.getElementById('bookBtn').innerText = 'Запись...';

        

        try {

            // Запрашиваем инфо о юзере у ВК (Имя, Фамилия, ID)

            const userData = await vkBridge.send('VKWebAppGetUserInfo').catch(() => ({id: 0, first_name: 'Клиент', last_name: ''}));

            

            const response = await fetch(`${N8N_URL}/create-booking`, {

                method: 'POST',

                headers: { 'Content-Type': 'application/json' },

                body: JSON.stringify({

                    vk_id: userData.id,

                    first_name: userData.first_name,

                    last_name: userData.last_name,

                    staff_id: staffId,

                    service_id: serviceId,

                    date: selectedDate,

                    time: selectedTime

                })

            });


            if (response.ok) {

                // Закрываем окно мини-апп, возвращаем человека в диалог

                vkBridge.send('VKWebAppClose', { status: 'success' });

            } else {

                alert('Ошибка создания записи.');

                document.getElementById('bookBtn').disabled = false;

            }

        } catch (error) {

            document.getElementById('bookBtn').disabled = false;

        }

    };

</script>
